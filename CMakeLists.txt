cmake_minimum_required(VERSION 3.24)
project(Sineweave VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
option(SINEWEAVE_BUILD_PLUGIN "Build JUCE VST3" ON)
option(SINEWEAVE_BUILD_TESTS "Build tests and benchmarks" ON)
option(SINEWEAVE_BUILD_LORIS_TOOLS "Build standalone upstream Loris CLI utilities" OFF)
option(SINEWEAVE_ENABLE_AVX2 "Include runtime-dispatched AVX2 kernel on Windows x64" ON)
set(SINEWEAVE_LORIS_PATH "" CACHE PATH "Local pinned Loris source")
set(SINEWEAVE_JUCE_PATH "" CACHE PATH "Local JUCE 8.0.15 source")
if(SINEWEAVE_LORIS_PATH)
  set(loris_SOURCE_DIR "${SINEWEAVE_LORIS_PATH}")
else()
  FetchContent_Declare(loris GIT_REPOSITORY https://github.com/kellyfitz/loris.git GIT_TAG eccd42ac847b08769643b0d0c36442ef0fd27f38 SOURCE_SUBDIR sineweave-integration-only)
  FetchContent_MakeAvailable(loris)
endif()
include(cmake/Loris.cmake)
add_library(sineweave_core STATIC src/core/Model.cpp src/core/Playback.cpp src/core/Synth.cpp)
if(MSVC AND CMAKE_SIZEOF_VOID_P EQUAL 8 AND SINEWEAVE_ENABLE_AVX2)
  target_sources(sineweave_core PRIVATE src/core/OscillatorAvx2.cpp)
  set_source_files_properties(src/core/OscillatorAvx2.cpp PROPERTIES COMPILE_OPTIONS /arch:AVX2)
  target_compile_definitions(sineweave_core PRIVATE SINEWEAVE_AVX2_AVAILABLE=1)
endif()
target_include_directories(sineweave_core PUBLIC src)
set_target_properties(sineweave_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(sineweave_analysis STATIC src/analysis/LorisAnalyser.cpp)
target_link_libraries(sineweave_analysis PUBLIC sineweave_core PRIVATE sineweave_loris)
set_target_properties(sineweave_analysis PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(SINEWEAVE_BUILD_TESTS)
  enable_testing()
  add_executable(loris_smoke tests/loris_smoke.cpp)
  target_link_libraries(loris_smoke PRIVATE sineweave_loris)
  add_test(NAME loris_smoke COMMAND loris_smoke)
  add_executable(core_tests tests/core_tests.cpp tests/AllocationProbe.cpp)
  target_link_libraries(core_tests PRIVATE sineweave_analysis sineweave_loris)
  add_test(NAME core_tests COMMAND core_tests)
  add_executable(synth_benchmark tests/benchmark.cpp)
  target_link_libraries(synth_benchmark PRIVATE sineweave_core)
  add_executable(trajectory_benchmark tests/trajectory_benchmark.cpp)
  target_link_libraries(trajectory_benchmark PRIVATE sineweave_core)
endif()
if(SINEWEAVE_BUILD_PLUGIN)
  if(SINEWEAVE_JUCE_PATH)
    add_subdirectory("${SINEWEAVE_JUCE_PATH}" "${CMAKE_BINARY_DIR}/juce")
  else()
    FetchContent_Declare(juce GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 91ad83ae34a81e0833b1a2b0866f54846370ae53 GIT_SHALLOW FALSE)
    FetchContent_MakeAvailable(juce)
  endif()
  juce_add_plugin(Sineweave COMPANY_NAME "Sineweave" BUNDLE_ID "org.sineweave.instrument"
    PLUGIN_MANUFACTURER_CODE Snwv PLUGIN_CODE Snw1 FORMATS VST3
    PRODUCT_NAME "Sineweave" IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE
    IS_MIDI_EFFECT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
  target_sources(Sineweave PRIVATE src/plugin/Processor.cpp src/plugin/Editor.cpp src/plugin/Worker.cpp)
  target_link_libraries(Sineweave PRIVATE sineweave_analysis juce::juce_audio_utils juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
  target_compile_definitions(Sineweave PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
  if(SINEWEAVE_BUILD_TESTS)
    add_executable(plugin_tests tests/plugin_tests.cpp tests/AllocationProbe.cpp)
    target_link_libraries(plugin_tests PRIVATE Sineweave)
    target_include_directories(plugin_tests PRIVATE $<TARGET_PROPERTY:Sineweave,INCLUDE_DIRECTORIES>)
    target_compile_definitions(plugin_tests PRIVATE $<TARGET_PROPERTY:Sineweave,COMPILE_DEFINITIONS>)
    add_test(NAME plugin_tests COMMAND plugin_tests)
  endif()
endif()
